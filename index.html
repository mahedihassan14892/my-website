// ৪. সিরিয়াল তালিকা রিয়েলটাইম শুনো (দূরত্ব ছাড়াই যেকোনো স্থান থেকে তালিকা দেখানোর ব্যবস্থা) onSnapshot(collection(db, "active_queue"), (snapshot) => { activeQueue = []; snapshot.forEach((docSnap) => { activeQueue.push({ id: docSnap.id, ...docSnap.data() }); }); activeQueue.sort((a, b) => new Date(a.entryTime) - new Date(b.entryTime)); const countEl = document.getElementById('queueCount'); if(countEl) countEl.innerText = activeQueue.length; // যেকোনো স্থান থেকেই যেন তালিকা সাথে সাথে স্ক্রিনে রেন্ডার হয় renderQueueUIWithoutDistanceRestriction(); }); // ৫. কিউ UI রেন্ডার করার ফাংশন (দূরত্ব ছাড়াই তালিকা দেখাবে) function renderQueueUIWithoutDistanceRestriction() { const container = document.getElementById('activeQueueList'); if(!container) return; if(activeQueue.length === 0) { container.innerHTML = '
বর্তমানে সিরিয়ালে কেউ নেই
'; return; } const now = new Date(); let html = ''; for (const rider of activeQueue) { const entry = new Date(rider.entryTime); const diffMs = now - entry; const totalSec = Math.floor(diffMs / 1000); const hrs = Math.floor(totalSec / 3600); const mins = Math.floor((totalSec % 3600) / 60); const secs = totalSec % 60; const timeStr = `${hrs > 0 ? hrs + 'ঘণ্টা ' : ''}${mins}মি. ${secs}সে.`; let breakBadge = ''; if(rider.breakStatus) { const breakStart = new Date(rider.breakStartTime); const elapsedMins = Math.floor((now - breakStart) / 60000); const maxAllowed = rider.breakStatus === 'prayer' ? 50 : 30; const remainingMins = maxAllowed - elapsedMins; if(remainingMins > 0) { breakBadge = `
${rider.breakStatus === 'prayer' ? 'নামাজের ব্রেক' : 'খাবারের ব্রেক'} (অবশিষ্ট: ${remainingMins} মি.)
`; } } html += `
${activeQueue.indexOf(rider) + 1}

${rider.name}

${rider.type === 'online' ? 'অনলাইন (অটো)' : 'অফলাইন (ম্যানুয়াল)'}

${rider.phone}

${breakBadge}

অপেক্ষার সময়

${timeStr}

`; } container.innerHTML = html; }
